```powershell
function Normalize-PathspecValue($value) {
  $p = $value.Trim().Trim('"').Replace('\', '/').TrimEnd('/').ToLowerInvariant()
  if ($p -eq '' -or $p.StartsWith('#')) { return $null }
  return $p
}
function Convert-GitStatusZ($text) {
  $records = $text -split "`0" | Where-Object { $_ }
  for ($i = 0; $i -lt $records.Count; $i++) {
    if ($records[$i].Length -lt 4) { throw "Malformed git status record: $($records[$i])" }
    $xy = $records[$i].Substring(0, 2)
    $path = Normalize-PathspecValue $records[$i].Substring(3)
    if ($xy -match '[RC]') {
      $source = if ($i + 1 -lt $records.Count) { Normalize-PathspecValue $records[++$i] } else { '<missing-source>' }
      [pscustomobject]@{ XY = $xy; Path = $path; Source = $source }
    } else {
      [pscustomobject]@{ XY = $xy; Path = $path; Source = $null }
    }
  }
}
$statusEntries = @(Convert-GitStatusZ (git status --porcelain=v1 -z -uall))
$unsafeStatus = $statusEntries | Where-Object {
  $_.XY -match '[DRCTU]' -or ($_.XY[0] -ne ' ' -and $_.XY -ne '??')
}
if ($unsafeStatus) {
  $unsafeStatus | ForEach-Object { "UNSAFE_STATUS $($_.XY) $($_.Path)" }
  exit 1
}
$statusPaths = $statusEntries.Path | Where-Object { $_ } | Sort-Object -Unique
$files = @(
  Get-ChildItem docs\audit\worktree-remediation\*.pathspec.txt
  Get-ChildItem docs\audit\worktree-remediation\*.hold.txt
) | Sort-Object FullName -Unique
$classifiers = foreach ($file in $files) {
  Get-Content $file.FullName | ForEach-Object { Normalize-PathspecValue $_ }
}
$classifiers = $classifiers | Where-Object { $_ } | Sort-Object -Unique
$uncovered = foreach ($path in $statusPaths) {
  $covered = $false
  foreach ($entry in $classifiers) {
    if ($path -eq $entry -or $path.StartsWith("$entry/")) { $covered = $true; break }
  }
  if (-not $covered) { $path }
}
if ($uncovered) { $uncovered } else { 'NO_UNSAFE_STATUS_STATES'; 'NO_UNCLASSIFIED_STATUS_PATHS' }
```

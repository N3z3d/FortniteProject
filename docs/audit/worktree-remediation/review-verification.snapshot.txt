```powershell
$problems = @()
$paths = Get-Content docs\audit\worktree-remediation\remediation-commit.pathspec.txt | Where-Object { $_ -and (Test-Path -LiteralPath $_) }
foreach ($p in $paths) {
  git ls-files --error-unmatch -- $p *> $null
  if ($LASTEXITCODE -ne 0) {
    $output = git diff --no-index --check -- /dev/null $p 2>&1
    $bad = $output | Where-Object { $_ -and ($_ -notmatch '^warning: in the working copy of ') }
    if ($bad) { $problems += ($bad | ForEach-Object { "$p :: $_" }) }
  }
}
if ($problems.Count -gt 0) { $problems; exit 1 } else { 'NO_WHITESPACE_ERRORS' }
```
